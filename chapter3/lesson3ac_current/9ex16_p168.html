លំហាត់ទី១៦៖ ប្រដាប់សាកថ្មទូរស័ព្ទ

ប្រដាប់សាកថ្មទូរស័ព្ទមានត្រង់ស្វូមួយដែលបញ្ចុះតង់ស្យុងឆ្លាស់ពី 240 V មក 5.0 V ដើម្បីសាកថ្មទូរស័ព្ទ 3.7 V ជាចរន្តជាប់ ។ ដើម្បីសាកថ្មទូរស័ព្ទបានគេត្រូវបំប្លែងវាពីចរន្តឆ្លាស់ទៅចរន្តជាប់ដោយប្រើស្ពានឌីយ៉ូត(មេរៀនទី 4 ក្រោយ)គឺបំប្លែងពី AC = 5.0 V ទៅជា DC = 5.0 V ។ សន្មតថា របុំមធ្យមមាន 30 ស្ពៀ និងប្រដាប់សាកថ្មផ្តល់ចរន្ត 2.00 A ។

គណនា ៖

  • ក. ចំនួនស្ពៀនៅរបុំបឋម
  • ខ. ចរន្តអគ្គិសនីនៅរបុំបឋម
  • គ. អានុភាពបញ្ជូន ។

គំនូសបំព្រួញ 2D: ដំណើរការប្រដាប់សាកថ្ម (Adapter)

ក

គណនាចំនួនស្ពៀនៅរបុំបឋម ($N_1$)

តាមរូបមន្តផលធៀបបំប្លែងនៃត្រង់ស្វូ ៖

$$\frac{V_1}{V_2} = \frac{N_1}{N_2}$$

ទាញរក $N_1$ ៖

$$N_1 = \frac{V_1 \cdot N_2}{V_2}$$

ដោយ ៖

  • $V_1 = 240 \text{ V}$ (តង់ស្យុងចូល)
  • $V_2 = 5.0 \text{ V}$ (តង់ស្យុងចេញពីត្រង់ស្វូ)
  • $N_2 = 30$ ស្ពៀ

ជំនួសលេខ ៖

$$N_1 = \frac{240 \times 30}{5.0} = \frac{7200}{5} = 1440$$

ដូចនេះ ចំនួនស្ពៀនៅរបុំបឋមគឺ $N_1 = 1440$ ស្ពៀ ។

ខ

គណនាចរន្តអគ្គិសនីនៅរបុំបឋម ($I_1$)

សន្មតថាជាត្រង់ស្វូបរិសុទ្ធ (គ្មានការខាតបង់ថាមពល) ៖

$$\frac{V_1}{V_2} = \frac{I_2}{I_1} \quad \text{ឬ} \quad V_1 \cdot I_1 = V_2 \cdot I_2$$

ទាញរក $I_1$ ៖

$$I_1 = \frac{V_2 \cdot I_2}{V_1}$$

ដោយ $V_1 = 240 \text{ V}, \ V_2 = 5.0 \text{ V}, \ I_2 = 2.00 \text{ A}$

ជំនួសលេខ ៖

$$I_1 = \frac{5.0 \times 2.00}{240} = \frac{10}{240} = \frac{1}{24} \approx 0.042 \text{ A}$$

ដូចនេះ ចរន្តអគ្គិសនីនៅរបុំបឋមគឺ $I_1 \approx 0.042 \text{ A}$ (ឬ $41.67 \text{ mA}$) ។

គ

គណនាអានុភាពបញ្ជូន ($P$)

អានុភាពបញ្ជូន (អានុភាពផ្តល់ដោយត្រង់ស្វូ) ៖

$$P = P_1 = P_2 = V_2 \cdot I_2$$

ជំនួសលេខ ៖

$$P = 5.0 \times 2.00 = 10 \text{ W}$$

ដូចនេះ អានុភាពបញ្ជូនគឺ $P = 10 \text{ W}$ ។
